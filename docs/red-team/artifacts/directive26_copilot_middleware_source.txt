/**
 * Edge-safe session verify for middleware.
 * Prefer jose jwtVerify (Edge-compatible) — custom Web Crypto previously failed closed
 * on valid HS256 cookies (demo/UAT), blocking Premium QA of /app/* surfaces.
 */
import { jwtVerify } from "jose";
import { NextRequest, NextResponse } from "next/server";

export const SESSION_COOKIE = "hvcg_copilot_session";

const PUBLIC_API_PREFIXES = [
  "/api/health",
  "/api/auth/demo",
  "/api/auth/session",
  "/api/auth/logout",
  "/api/auth/start",
];

function secretKey() {
  const raw =
    process.env.COPILOT_SESSION_SECRET ||
    process.env.NEXTAUTH_SECRET ||
    "hvcg-copilot-dev-only-session-secret-change-me";
  return new TextEncoder().encode(raw);
}

type SessionClaims = {
  exp?: number;
  organizationId?: string;
  assessmentId?: string;
  workspaceId?: string;
  userId?: string;
  role?: string;
};

async function verifyJwtHs256(token: string | undefined | null): Promise<SessionClaims | null> {
  if (!token) return null;
  try {
    const { payload } = await jwtVerify(token, secretKey(), { algorithms: ["HS256"] });
    const claims = payload as unknown as SessionClaims;
    if (!claims.userId || !claims.organizationId || !claims.assessmentId || !claims.workspaceId) {
      return null;
    }
    if (typeof claims.exp === "number" && claims.exp * 1000 < Date.now()) return null;
    return claims;
  } catch {
    return null;
  }
}

function isPublicApiPath(pathname: string) {
  return PUBLIC_API_PREFIXES.some((p) => pathname === p || pathname.startsWith(`${p}/`));
}

function isProtectedPagePath(pathname: string) {
  return (
    pathname === "/app" ||
    pathname.startsWith("/app/") ||
    pathname === "/admin" ||
    pathname.startsWith("/admin/") ||
    pathname.startsWith("/onboarding/") ||
    pathname.startsWith("/assessment/") ||
    pathname.startsWith("/mri/") ||
    pathname === "/proposals" ||
    pathname === "/schedule" ||
    pathname === "/consent"
  );
}

export async function middleware(req: NextRequest) {
  const { pathname } = req.nextUrl;
  const token = req.cookies.get(SESSION_COOKIE)?.value;

  if (isProtectedPagePath(pathname)) {
    const claims = await verifyJwtHs256(token);
    if (!claims) {
      const login = new URL("/login", req.url);
      login.searchParams.set("next", pathname);
      return NextResponse.redirect(login);
    }
    if (pathname.startsWith("/admin")) {
      const role = claims.role;
      if (role !== "hvcg_admin" && role !== "hvcg_consultant") {
        return NextResponse.redirect(new URL("/app", req.url));
      }
    }
    return NextResponse.next();
  }

  if (pathname.startsWith("/api/") && !isPublicApiPath(pathname)) {
    const claims = await verifyJwtHs256(token);
    if (!claims) {
      return NextResponse.json(
        { error: "Authentication required", code: "UNAUTHENTICATED" },
        { status: 401 }
      );
    }
  }

  return NextResponse.next();
}

export const config = {
  matcher: [
    "/app/:path*",
    "/admin/:path*",
    "/onboarding/:path*",
    "/assessment/:path*",
    "/mri/:path*",
    "/proposals",
    "/schedule",
    "/consent",
    "/api/:path*",
  ],
};
