import { NextRequest, NextResponse } from "next/server";
import { AuthError, authErrorResponse, requireTenantContext } from "@/lib/auth/guards";
import { readSessionFromRequest } from "@/lib/auth/session";
import {
  buildPreCallBrief,
  buildQualificationHints,
  normalizeAttribution,
  toIntegrationPreCallBrief,
} from "@/lib/gtm/enrichment";
import { readStore } from "@/lib/store";

/** GTM enrichment — producer only; does not orchestrate campaigns or write CRM. */
export async function GET(req: NextRequest) {
  try {
    const session = await readSessionFromRequest(req);
    if (!session) throw new AuthError("UNAUTHENTICATED", "Authentication required", 401);
    const store = readStore(session.workspaceId);
    const { assessment, organization } = requireTenantContext(store, session);
    const attributionAudit = store.audit.find(
      (a) => a.action === "gtm.attribution.capture" && a.target === assessment.id
    );
    const meta = (attributionAudit?.metadata || {}) as Record<string, unknown>;
    const attribution = normalizeAttribution({
      campaignId: typeof meta.campaignId === "string" ? meta.campaignId : undefined,
      sourceAttribution:
        typeof meta.sourceAttribution === "string" ? meta.sourceAttribution : undefined,
      utm: meta.utm as never,
    });
    const brief = buildPreCallBrief({ organization, assessment, attribution });
    const qualification = buildQualificationHints(assessment);
    return NextResponse.json({
      contractVersion: "gtm.assessment-enrichment.v1",
      liveOrchestration: false,
      commercialBinding: false,
      observationOnly: true,
      commercialAuthority: "revenue-os",
      liveDispatch: false,
      attribution,
      qualificationHints: qualification,
      preCallBrief: brief,
      integrationPreCallBrief: null,
      note: "Producer enrichment only. Map via toIntegrationPreCallBrief when a bookingId exists. Not CRM/GTM orchestration.",
    });
  } catch (err) {
    return authErrorResponse(err);
  }
}

export async function POST(req: NextRequest) {
  try {
    const session = await readSessionFromRequest(req);
    if (!session) throw new AuthError("UNAUTHENTICATED", "Authentication required", 401);
    const body = await req.json().catch(() => ({}));
    const store = readStore(session.workspaceId);
    const { assessment, organization } = requireTenantContext(store, session);
    const attribution = normalizeAttribution({
      campaignId: body.campaignId || body.campaign,
      sourceAttribution: body.sourceAttribution,
      source: body.source,
      utm: body.utm,
    });
    const brief = buildPreCallBrief({ organization, assessment, attribution });
    const bookingId = typeof body.bookingId === "string" ? body.bookingId : undefined;
    return NextResponse.json({
      ok: true,
      liveOrchestration: false,
      commercialBinding: false,
      observationOnly: true,
      commercialAuthority: "revenue-os",
      liveDispatch: false,
      attribution,
      qualificationHints: buildQualificationHints(assessment),
      preCallBrief: brief,
      integrationPreCallBrief: bookingId
        ? toIntegrationPreCallBrief(brief, {
            bookingId,
            atlasClientCode: typeof body.atlasClientCode === "string" ? body.atlasClientCode : undefined,
          })
        : null,
    });
  } catch (err) {
    return authErrorResponse(err);
  }
}
