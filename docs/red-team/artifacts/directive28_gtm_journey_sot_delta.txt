+function mapExperimentStatus(status: GtmExperiment['status']): ExperimentSpecV1['status'] {
+  switch (status) {
+    case 'promoted':
+    case 'learned':
+    case 'measured':
+    case 'attributed':
+      return 'completed';
+    case 'experimenting':
+      return 'running';
+    case 'rolled_back':
+      return 'abandoned';
+    case 'observed':
+    case 'analyzed':
+    case 'hypothesized':
+    case 'variant_generated':
+    case 'qa_passed':
+    case 'compliance_checked':
+    default:
+      return 'draft';
+  }

+/**
+ * Emit booking-event.v1 from an existing SYN-GTM dry-run booking.
+ * Refuses non-dry-run bookings (liveDispatch impossible on this path).
+ */
+export function toBookingEventV1(input: {
+  booking: Extract<BookingResult, { ok: true }>;
+  leadId: string;
+  contactEmail: string;
+  campaignId: string;
+  correlationId?: string;
+}): BookingEventV1 {
+  if (input.booking.dryRun !== true) {
+    throw new Error('booking_event_requires_dryRun_true — liveDispatch impossible');
+  }
+  const bookingId = input.booking.meetingId;
+  const now = new Date().toISOString();
+  return bookingEventV1Schema.parse({
+    contractVersion: BOOKING_EVENT_CONTRACT_VERSION,
+    bookingId,
+    envelope: {
+      idempotencyKey: `booking|${bookingId}`,
+      sourceSystem: '360',
+      destinationSystem: 'atlas',
+      entity: 'booking',
+      operation: 'stage',
+      version: BOOKING_EVENT_CONTRACT_VERSION,
+      replaySemantics: 'return-existing',
+      trace: {
+        correlationId: input.correlationId ?? bookingId,
+        sourceSystem: '360',
+        destinationSystem: 'atlas',
+        eventId: `evt-book-${bookingId}`,
+        entityId: bookingId,
+        campaignId: input.campaignId,
+        timestamp: now,
+        version: BOOKING_EVENT_CONTRACT_VERSION,
+        outcome: 'accepted',
+      },
+      actor: {
+        identityClass: 'system_service',
+        principalId: '360-gtm-synthetic',
+        scopes: ['booking:stage'],
+      },
+    },
+    leadRef: {
+      system: '360',
+      entity: 'lead',
+      id: input.leadId,
+    },
+    contactEmail: input.contactEmail,
+    startsAt: input.booking.slot.start,
+    endsAt: input.booking.slot.end,
+    status: input.booking.status === 'confirmed' ? 'confirmed' : 'requested',
+    meetingProvider: 'microsoft_calendar_mock',
+    attribution: {
+      source: 'SYN-GTM',
