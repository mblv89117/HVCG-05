 * GCC-RT-07: accept either
 * 1) valid Atlas HMAC service attestation, or
 * 2) authenticated platform_admin session.
 * Unsigned machine POSTs are rejected.
 */
export async function POST(req: Request) {
  try {
    const rawBody = await req.text();
    const attestation = verifyAtlasHandoffAttestation({
      rawBody,
      timestampHeader: req.headers.get("x-atlas-gcc-timestamp"),
      signatureHeader: req.headers.get("x-atlas-gcc-signature"),
    });

    let authMode: "hmac" | "platform_admin";
    if (attestation.ok) {
      authMode = "hmac";
    } else {
      try {
        await requirePlatformAdminAccess();
        authMode = "platform_admin";
      } catch {
        return NextResponse.json(
          {
            error: "handoff_attestation_required",
            reason: attestation.reason,
            hint: "Provide X-Atlas-Gcc-Timestamp + X-Atlas-Gcc-Signature HMAC, or authenticate as platform_admin",
          },
          { status: 401 },
        );
      }
    }

    let body: AtlasGccActivationHandoff;
    try {
      body = JSON.parse(rawBody) as AtlasGccActivationHandoff;
    } catch {
      throw new AuthError("Invalid JSON body", 400);
    }

    const issues = assertAtlasGccActivationContract(body);
    if (issues.length > 0) {
      return NextResponse.json({ error: "invalid_contract", issues }, { status: 400 });
    }
    if (body.governance.autoProvisionAccess !== false) {
      return NextResponse.json({ error: "auto_provision_forbidden" }, { status: 400 });
    }

    const staged = stageAtlasGccActivation(body);
    return NextResponse.json({
      ok: true,
      created: staged.created,
      replay: !staged.created,
      provisioned: false,
      authMode,
      event: staged,
    });
  } catch (error) {
    return authErrorResponse(error);
  }
}

  workspaceId?: string;

export function newWorkspaceId(): string {
  return randomUUID();
}

/** Fail closed: only UUID workspace ids are accepted (path-traversal safe). */
export function assertSafeWorkspaceId(workspaceId: string): string {
  const id = String(workspaceId || "").trim();
  if (!/^[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(id)) {
    throw new Error("INVALID_WORKSPACE_ID");
  }
  return id.toLowerCase();
}

function workspaceFile(workspaceId: string): string {
  const id = assertSafeWorkspaceId(workspaceId);
  return path.join(WORKSPACES_DIR, `${id}.json`);
}

function emptyStore(workspaceId?: string): AppStore {
  return {
    organizations: [],
    users: [],
    assessments: [],
    integrations: [],
    documents: [],
    captures: [],
    consents: [],
    audit: [],
    pricing: DEFAULT_PRICING,
    salesOpportunities: [],
    atlasHandoffs: [],
    atlasLeadHandoffs: [],
    workspaceId,
  };
}

function normalizeStore(raw: Partial<AppStore>, workspaceId?: string): AppStore {
  const base = emptyStore(workspaceId);
  return {
    ...base,
    ...raw,
    organizations: raw.organizations || [],
    users: raw.users || [],
    assessments: raw.assessments || [],
    integrations: raw.integrations || [],
    documents: raw.documents || [],
    captures: raw.captures || [],
    consents: raw.consents || [],
    audit: raw.audit || [],
    pricing: raw.pricing?.length ? raw.pricing : DEFAULT_PRICING,
    salesOpportunities: raw.salesOpportunities || [],
    atlasHandoffs: raw.atlasHandoffs || [],
    atlasLeadHandoffs: raw.atlasLeadHandoffs || [],
    workspaceId: workspaceId || raw.workspaceId,
    session: raw.session
      ? {
          userId: raw.session.userId,
          organizationId: raw.session.organizationId,
          mode: raw.session.mode === "demo" ? "demo" : "uat",
          sampleData: true,
        }
      : undefined,
  };
}

function ensureDirs() {
  if (!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR, { recursive: true });
  if (!fs.existsSync(WORKSPACES_DIR)) fs.mkdirSync(WORKSPACES_DIR, { recursive: true });
}

/**
 * Read a per-session durable workspace store.
 * COPILOT-RT-02: never falls back to a shared global store.json.
 */
export function readStore(workspaceId: string): AppStore {
  ensureDirs();
  const file = workspaceFile(workspaceId);
  if (!fs.existsSync(file)) {
    return emptyStore(assertSafeWorkspaceId(workspaceId));
  }
  const raw = fs.readFileSync(file, "utf8");
  return normalizeStore(JSON.parse(raw) as Partial<AppStore>, assertSafeWorkspaceId(workspaceId));
}

/**
 * Persist a per-session durable workspace store.
 * Refuses to write the legacy global path.
 */
export function writeStore(store: AppStore, workspaceId: string) {
  ensureDirs();
  const id = assertSafeWorkspaceId(workspaceId);
  const file = workspaceFile(id);
  if (path.resolve(file) === path.resolve(LEGACY_GLOBAL_STORE_FILE)) {
    throw new Error("GLOBAL_STORE_WRITE_FORBIDDEN");
  }
  const next = normalizeStore({ ...store, workspaceId: id }, id);
  fs.writeFileSync(file, JSON.stringify(next, null, 2));
}

export function updateStore(mutator: (store: AppStore) => void, workspaceId: string): AppStore {
  const store = readStore(workspaceId);
  mutator(store);
  writeStore(store, workspaceId);
  return store;
}

export function workspaceExists(workspaceId: string): boolean {

  } catch {
    return false;
  }
}

export async function POST(req: NextRequest) {
  const body = await req.json().catch(() => ({}));
  const campaignId =
    typeof body.campaignId === "string"
      ? body.campaignId
      : typeof body.campaign === "string"
        ? body.campaign

  const sourceAttribution =
    typeof body.sourceAttribution === "string"
      ? body.sourceAttribution

    contactEmail: String(body.contactEmail || body.email || "uat@example.invalid"),

    website: body.website ? String(body.website) : undefined,
    jobTitle: body.jobTitle ? String(body.jobTitle) : undefined,
    employeeCountRange: body.employeeCountRange ? String(body.employeeCountRange) : undefined,
    industry: body.industry ? String(body.industry) : undefined,
    primaryObjective: body.primaryObjective ? String(body.primaryObjective) : undefined,
  });
  store.workspaceId = workspaceId;

  addAudit(store, {
    organizationId: store.organizations[0].id,
    actorId: store.users[0].id,
    action: "gtm.attribution.capture",
    target: store.assessments[0].id,
    metadata: {
      campaignId,
      sourceAttribution,
      workspaceId,
      utm: {
        source: body.utm_source,
        medium: body.utm_medium,
        campaign: body.utm_campaign,
        content: body.utm_content,
        term: body.utm_term,
      },
    },
  });
  writeStore(store, workspaceId);

  const { token, session } = await signSession({
    userId: store.users[0].id,
    organizationId: store.organizations[0].id,
    assessmentId: store.assessments[0].id,
    workspaceId,
    mode: "uat",
    role: store.users[0].role,
  });

  const res = NextResponse.json({
    started: true,
    workspaceId,
    assessment: store.assessments[0],
    organization: store.organizations[0],
    mode: session.mode,
    sampleData: true,
    productionClientDataAllowed: false,
    attribution: { campaignId, sourceAttribution },
  });
  applySessionCookie(res, token);
  return res;

/**
 * GCC-RT-07: accept either
 * 1) valid Atlas HMAC service attestation, or
 * 2) authenticated platform_admin session.
 * Unsigned machine POSTs are rejected.
 */
export async function POST(req: Request) {
  try {
    const rawBody = await req.text();
    const attestation = verifyAtlasHandoffAttestation({
      rawBody,
      timestampHeader: req.headers.get("x-atlas-gcc-timestamp"),
      signatureHeader: req.headers.get("x-atlas-gcc-signature"),
    });

    let authMode: "hmac" | "platform_admin";
    if (attestation.ok) {
      authMode = "hmac";
    } else {
      try {
        await requirePlatformAdminAccess();
        authMode = "platform_admin";
      } catch {
        return NextResponse.json(
          {
            error: "handoff_attestation_required",
            reason: attestation.reason,
            hint: "Provide X-Atlas-Gcc-Timestamp + X-Atlas-Gcc-Signature HMAC, or authenticate as platform_admin",
          },
          { status: 401 },
        );
      }
    }

    let body: AtlasGccActivationHandoff;
    try {
      body = JSON.parse(rawBody) as AtlasGccActivationHandoff;
    } catch {
      throw new AuthError("Invalid JSON body", 400);
    }

    const issues = assertAtlasGccActivationContract(body);
    if (issues.length > 0) {
      return NextResponse.json({ error: "invalid_contract", issues }, { status: 400 });
    }
    if (body.governance.autoProvisionAccess !== false) {
      return NextResponse.json({ error: "auto_provision_forbidden" }, { status: 400 });
    }

    const staged = stageAtlasGccActivation(body);
    return NextResponse.json({
      ok: true,
      created: staged.created,
      replay: !staged.created,
      provisioned: false,
      authMode,
      event: staged,
    });
  } catch (error) {
    return authErrorResponse(error);
  }
}
